JavaScript
// 在 setup() 内部添加
const onTouchMove = (e) => {
    // 阻止屏幕滚动，确保拖拽流畅
    if (draggedCard.value) e.preventDefault();
};

const onTouchEnd = (e, card) => {
    if (!draggedCard.value) return;

    // 获取手指离开屏幕时的坐标
    const touch = e.changedTouches[0];
    // 关键：根据坐标找到下方的元素
    const element = document.elementFromPoint(touch.clientX, touch.clientY);
    
    // 寻找最近的包含 data-area 属性的父元素
    const areaElement = element?.closest('[data-area]');
    
    if (areaElement) {
        const areaName = areaElement.getAttribute('data-area');
        onDrop(areaName);
    } else {
        // 如果没丢进目标区，重置状态
        draggedCard.value = null;
        isDraggingOver.value = null;
    }
};

// 在 return 中暴露
return {
    // ... 其他变量
    onTouchMove, onTouchEnd,
    // ...
}

async function downloadImage(url, filename) {
    const filePath = path.join(IMAGE_DIR, filename);
    if (fs.existsSync(filePath)) return;
    try {
        const response = await axios({
            url: url,
            method: 'GET',
            responseType: 'stream',
            headers: {
                'Referer': 'https://fecipher.card.moe/',
                'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36'
            }
        });
        const writer = fs.createWriteStream(filePath);
        response.data.pipe(writer);
        return new Promise((resolve, reject) => {
            writer.on('finish', resolve);
            writer.on('error', reject);
        });
    } catch (e) {
        console.error(`下载失败 ${filename}: ${e.message}`);
    }
}

async function downloadImage(url, filename) {
    const filePath = path.join(IMAGE_DIR, filename);
    if (fs.existsSync(filePath)) return; // 跳过已存在的图片

    try {
        const response = await axios({
            url: url,
            method: 'GET',
            responseType: 'stream',
            headers: {
                'Referer': 'https://fecipher.card.moe/',
                'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36'
            }
        });

        const writer = fs.createWriteStream(filePath);
        response.data.pipe(writer);
        return new Promise((resolve, reject) => {
            writer.on('finish', resolve);
            writer.on('error', reject);
        });
    } catch (e) {
        console.error(`下载失败 ${filename}: ${e.message}`);
    }
}

/**
 * 提取关键字辅助函数
 */
function extractKeywords(text) {
    if (!text) return [];
    const matches = text.match(/【(.*?)】/g);
    return matches ? matches.map(m => m.replace(/[【】]/g, '')) : [];
}

 * 解析技能消耗
 */
function parseSkillCost(html) {
    if (!html) return 0;
    const bondMatch = html.match(/翻面(\d+)个/);

在《Fire Emblem Cipher》（圣火降魔录：各色纹章，简称 FE0）中，一个完整的游戏流程由多个轮替的**回合（Turn）**组成。每个回合内部遵循严格的阶层式结构。

以下是详细的游戏阶段划分及玩家在各阶段的操作指南：

1. 开始阶段 (Beginning Phase)
这是回合的准备时期，主要用于状态重置。

解除横置（Ready）： 将你战场上所有处于“横置（被消耗）”状态的卡牌转回竖直状态。

抽卡（Draw）： 从牌组最上方抽 1 张牌加入手牌。（注意：先攻玩家的第一回合通常不抽牌）。

触发效果： 处理所有标有“回合开始时”的技能。

2. 羁绊阶段 (Bond Phase)
羁绊（Bond）是游戏中的能量系统。

放置羁绊： 玩家可以从手牌中选择最多 1 张卡牌，正面向上（或根据效果反面向上）放置到羁绊区。

重要规则： 一旦放置，本回合无法更改。羁绊的数量决定了你能出击多强大的英雄。

3. 出击阶段 (Deployment Phase)
这是你构筑战场战力的核心阶段。

出击（Deploy）： 支付等同于卡牌“出击费用”的羁绊数量（横置或检查数量，取决于具体规则），将手牌中的卡片放置到战场（前卫或后卫）。

晋升/转职（Promote）： 如果战场上已有同名卡，你可以支付“转职费用”，将高阶卡叠在原有卡片上方。这通常会触发“抽牌”奖励并增强战力。

移动（Move）： 消耗 1 个羁绊，手动将战场上的一张单位从前卫移至后卫，或反之。

使用技能： 玩家可以发动标记为“【起】”的动作阶段技能（需支付相应代价）。

4. 攻击阶段 (Attack Phase)
这是最激烈的博弈阶段，包含多个子步骤：

指定攻击： 选择一名“竖直状态”的单位，将其横置，并宣布攻击目标（对方前卫或特定射程内的后卫）。

支援步骤（Support）：

攻击方和被攻击方同时从牌组顶翻开一张卡放入“支援区”。

计算战力： 最终战力 = 基础战力 + 支援卡提供的支援力 + 技能加成。

支援技能： 触发支援卡上的“【支】”纹章效果（如攻击纹章、防御纹章）。

胜负判定： * 如果攻击方战力 ≥ 防御方战力，则攻击成功。

多次攻击： 只要你还有处于竖直状态且未攻击过的单位，就可以重复上述流程。

5. 结束阶段 (End Phase)
清理支援： 将支援区的所有卡片送往弃牌堆。

效果失效： 所有“本回合内有效”的临时增益或减益宣告结束。

移交回合： 宣布回合结束，轮到对手。
